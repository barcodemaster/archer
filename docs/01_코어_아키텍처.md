## Singleton\<T\> - 제네릭 싱글톤 베이스

> `Assets/Scripts/Manager/Singleton.cs`

```csharp
using UnityEngine;
using UnityEngine.SceneManagement;                              // ⑤ 씬 로드 이벤트용

public class Singleton<T> : MonoBehaviour where T : MonoBehaviour
{
    private static T _instance;
    private static bool _destroyed;                              // ① 앱 종료 시 재생성 방지 플래그

    public static T Instance
    {
        get
        {
            if (_destroyed)                                      // ①
                return null;

            if (_instance == null)
            {
                _instance = (T)FindAnyObjectByType(typeof(T));

                if (_instance == null)
                {
                    // 없으면 새 GameObject를 생성하여 컴포넌트 부착
                    GameObject go = new GameObject($"@{typeof(T).Name}");  // '@' 접두사로 매니저 구분
                    _instance = go.AddComponent<T>();
                }

                DontDestroyOnLoad(_instance.gameObject);
            }

            return _instance;
        }
    }

    protected virtual void Awake()                               // ②
    {
        if (_instance != null && _instance != this)
        {
            Destroy(gameObject);
            return;
        }

        _instance = this as T;
        _destroyed = false;
        DontDestroyOnLoad(gameObject);
    }

    protected virtual void OnDestroy()                           // ③
    {
        if (_instance == this)
        {
            _instance = null;
            _destroyed = true;
            SceneManager.sceneLoaded += OnSceneLoaded;
        }
    }

    protected virtual void OnApplicationQuit()                   // ④
    {
        _destroyed = true;
    }

    private static void OnSceneLoaded(Scene scene, LoadSceneMode mode)  // ⑤
    {
        _destroyed = false;
        SceneManager.sceneLoaded -= OnSceneLoaded;
    }
}
```

#### 코드 해설
| # | 설명 |
|---|------|
| ① | `_destroyed` 플래그로 앱 종료 시 파괴된 싱글톤의 재생성을 방지한다. Unity는 `OnDestroy` 호출 순서를 보장하지 않으므로, 이미 파괴된 싱글톤을 다른 `OnDestroy`에서 접근하면 새 인스턴스가 생성되는 버그가 발생한다. `_destroyed`가 `true`이면 `Instance`가 `null`을 반환하여 이를 차단한다 |
| ② | `Awake()`에서 중복 인스턴스를 제거한다. 씬에 직접 배치한 인스턴스와 `Instance` getter의 자동 생성이 충돌할 때, 후발 인스턴스를 `Destroy`한다. `protected virtual`이므로 하위 클래스에서 `override`하여 초기화 로직을 추가할 수 있다 |
| ③ | `OnDestroy()`에서 인스턴스 참조를 정리하고 `_destroyed = true`로 즉시 재생성을 차단한다. 동시에 `SceneManager.sceneLoaded` 이벤트를 등록하여, 새 씬이 로드되면 `_destroyed`를 `false`로 복구한다 |
| ④ | `OnApplicationQuit()`에서 `_destroyed = true`를 설정하여, 종료 과정에서 다른 스크립트의 `OnDestroy`가 싱글톤에 접근해도 재생성되지 않도록 한다 |
| ⑤ | `OnSceneLoaded()`에서 `_destroyed`를 `false`로 복구하고, 이벤트 핸들러를 즉시 해제하여 메모리 누수를 방지한다. 이로써 씬 전환 후에는 필요 시 싱글톤이 재생성될 수 있다 |

### 설계 의도

> **왜 Lazy Initialization인가?** Unity에서 싱글톤을 구현하는 방법은 크게 세 가지다: (1) 씬에 미리 배치, (2) 프리팹으로 관리, (3) Lazy Init. 씬 배치는 씬마다 수동 배치가 필요하고, 프리팹 방식은 Resources 폴더 의존성이 생긴다. Lazy Init은 **어떤 씬에서든 최초 접근 시 자동 생성**되므로, 초기화 순서를 신경 쓸 필요가 없고 프리팹 의존성도 없다. 다만 `FindAnyObjectByType`으로 기존 인스턴스를 먼저 탐색하여, 에디터에서 수동 배치한 매니저도 존중한다.

> **왜 `_destroyed` 플래그가 필요한가?** Unity 앱 종료 시 `OnDestroy` 호출 순서가 보장되지 않는다. 예를 들어, AudioManager의 `OnDestroy`가 먼저 호출된 뒤 다른 스크립트의 `OnDestroy`에서 `AudioManager.Instance`에 접근하면, Lazy Init에 의해 새 AudioManager가 생성되어 콘솔에 경고가 출력된다. `_destroyed` 플래그는 이 시점에 `null`을 반환하여 불필요한 재생성을 차단한다. 씬 전환 시에는 `OnSceneLoaded`에서 플래그를 복구하여 정상적인 재생성을 허용한다.

- **Lazy Initialization**: `Instance` 프로퍼티 접근 시점에 초기화하여, 명시적 초기화 순서 관리 불필요
- **Auto-Create**: 씬에 배치하지 않아도 자동 생성 → 프리팹 의존성 제거
- **DontDestroyOnLoad**: 씬 전환 시에도 상태 유지
- **이름 규칙**: `@GameManager` 형태로 Hierarchy에서 매니저 오브젝트를 쉽게 식별
- **중복 방지**: `Awake()`에서 후발 인스턴스를 자동 파괴하여 씬 배치 + 코드 생성 충돌 해소
- **종료 안전**: `_destroyed` 플래그 + `OnApplicationQuit()`으로 앱 종료 시 재생성 버그 차단
- **씬 전환 복구**: `OnSceneLoaded()`로 새 씬 로드 후 재생성을 다시 허용하되, 이벤트 즉시 해제로 메모리 누수 방지

## GameManager - 글로벌 게임 상태

> `Assets/Scripts/Manager/GameManager.cs`

```csharp
using UnityEngine;
using UnityEngine.U2D;

public class GameManager : Singleton<GameManager>
{
    // 조이스틱 입력 방향 (UI_Joystick에서 설정, PlayerController에서 소비)
    public Vector2 JoystickDir { get; set; } = Vector2.zero;

    // 일시정지 상태 (레벨업, 장비창 등에서 true)
    public bool IsPaused { get; set; } = false;

    /// <summary>
    /// Singleton.Awake()를 override하여 리소스 프리로드를 수행한다.
    /// base.Awake() 호출로 중복 인스턴스 제거 + DontDestroyOnLoad을 보장한다.
    /// </summary>
    protected override void Awake()                              // ①
    {
        base.Awake();
        PreloadResources();
    }

    /// <summary>
    /// 모바일 첫 프레임 스터터 방지를 위해 주요 리소스를 미리 로드한다.
    /// </summary>
    private void PreloadResources()                              // ②
    {
        LoadResource<TextAsset>("Data/EquipmentData");
        LoadResource<TextAsset>("Data/UpgradeData");
        LoadResource<GameObject>("Prefabs/DamageText");
        LoadResource<GameObject>("Prefabs/LevelUpText");
        LoadResource<SpriteAtlas>("UI/Icon");
        LoadResource<SpriteAtlas>("UI/Common");
    }

    private void LoadResource<T>(string path) where T : Object
    {
        T resource = Resources.Load<T>(path);
        if (resource == null)
            Logger.LogWarning("GameManager", $"Failed to preload resource: {path}");
    }

    /// <summary>
    /// 부활 흐름을 제어한다. UI(조이스틱 표시)와 게임 로직(플레이어 부활)을 조율한다.
    /// </summary>
    public void RevivePlayer()                                   // ③
    {
        UIManager.Instance.ShowJoystick();
        PlayerController player = PlayerController.Instance;
        if (player != null)
            player.Revive();
    }
}
```

#### 코드 해설
| # | 설명 |
|---|------|
| ① | `protected override void Awake()`로 Singleton의 Awake를 override한다. 반드시 `base.Awake()`를 호출하여 중복 인스턴스 제거와 `DontDestroyOnLoad` 설정을 보장한다 |
| ② | `PreloadResources()`에서 게임에 필수적인 리소스를 미리 로드한다. 모바일에서 `Resources.Load` 첫 호출 시 I/O 지연으로 프레임 드랍이 발생하는데, 시작 시 미리 로드하면 런타임 스터터를 방지할 수 있다. 로드 실패 시 크래시 대신 `Logger`로 경고만 출력하여 Graceful Degradation을 구현한다 |
| ③ | `RevivePlayer()`에서 UI(조이스틱 복원)와 게임 로직(플레이어 부활)을 한 번에 조율한다. GameManager가 여러 시스템 간의 흐름을 중재하는 Mediator 역할을 담당한다 |

### 설계 의도

> **왜 이렇게 최소한의 역할만 갖는가?** 많은 Unity 프로젝트에서 GameManager가 수백 줄짜리 "God Object"가 되는 것이 흔한 안티패턴이다. 이 프로젝트에서는 GameManager를 **입력 중개, 일시정지 플래그, 리소스 프리로드, 게임 흐름 제어**만 담당하는 경량 허브로 설계했다. 스테이지 관리는 StageManager, 경험치는 ExpManager 등 각 책임을 별도 매니저로 분리했기 때문에, GameManager가 비대해지지 않으며 각 시스템을 독립적으로 수정할 수 있다.

> **왜 리소스를 프리로드하는가?** 모바일에서 `Resources.Load`의 첫 호출은 디스크 I/O를 동반하여 프레임 드랍을 일으킨다. 게임 시작 시 `PreloadResources()`로 주요 리소스를 미리 로드하면, 실제 게임 플레이 중 첫 프레임 스터터를 방지할 수 있다. 제네릭 `LoadResource<T>` 헬퍼로 로드 + 실패 로깅을 통일한다.

- **경량 허브**: 입력 방향, 일시정지 플래그, 리소스 프리로드, 부활 흐름 제어만 담당
- **중개 역할**: UI_Joystick → GameManager → PlayerController 데이터 흐름의 중간 허브
- **Pause 제어**: `IsPaused`를 체크하여 PlayerController, MonsterBase 등이 동작을 멈춤
- **리소스 프리로드**: Awake 시점에 주요 리소스를 미리 로드하여 모바일 첫 프레임 스터터 방지
- **게임 흐름 제어**: `RevivePlayer()`로 UI와 게임 로직 간의 중재자(Mediator) 역할 수행
- **Singleton 연계**: `base.Awake()` 호출로 중복 인스턴스 제거 + DontDestroyOnLoad 보장

### 데이터 흐름

## Define.cs - 열거형 및 상수 정의

> `Assets/Scripts/Utils/Define.cs`

```csharp
using System.Collections.Generic;
using UnityEngine;

public class Define
{
    // ─── 타일 타입 ───
    public enum ETileType
    {
        Path  = 0,   // 일반 통행로
        Wall  = 1,   // 벽 (기본 차단)
        Water = 2,   // 물 (WaterWalk/Fly만 통과)
        Spike = 3,   // 가시 (통과 가능하나 데미지)
    }

    // ─── 타일 통과 플래그 (비트 플래그) ───
    [System.Flags]
    public enum ETilePassFlag
    {
        None      = 0,
        Walk      = 1 << 0,   // ① 도보 이동 (Path, Spike)
        Fly       = 1 << 1,   // 비행 (모든 타일)
        WaterWalk = 1 << 2,   // 수상 보행 (Water)
        WallPass  = 1 << 3,   // 벽 통과 (Wall + Block)
    }

    // ─── 투사체 이동 방식 ───
    public enum EProjectileMoveType
    {
        Straight,   // 직선 이동 (기본)
        Arc,        // 포물선 (Nepenthes)
        Piercing,   // 관통 (직선 + 다중 히트)
    }

    // ─── 엔티티 상태 ───
    public enum EState
    {
        None,
        Idle,
        Move,
        Attack,
        Die,
        Jump,      // GhostMonster, TreantMinion 점프
        Skill1,    // ImpMischief 보스 스킬
        Skill2,
        Skill3,
    }

    // ─── 업그레이드 타입 (22개) ───
    public enum EUpgradeType
    {
        AttackSpeed,      // 공격속도
        FrontArrow,       // 전방 화살 추가
        MultiShot,        // 멀티 샷 (연속 발리)
        Piercing,         // 관통
        Headshot,         // 헤드 샷 (즉사)
        DiagonalArrow,    // 사선 화살 (±45°)
        WallBounce,       // 벽 반사
        SideArrow,        // 측면 화살 (±90°)
        Ricochet,         // 리코셰 (적간 도탄)
        BackArrow,        // 후방 화살 (180°)
        CriticalMaster,   // 크리티컬 마스터
        AttackBoost,      // 공격 부스트
        FastGrowth,       // 빠른 성장 (최대 레벨↑)
        HpBoost,          // HP 부스트
        WallPass,         // 장애물 통과
        WaterWalker,      // 워터 워커
        Dwarf,            // 난쟁이 (축소+크리티컬)
        Giant,            // 거인 (확대+공격력)
        Rage,             // 분노 (낮은 HP → 높은 공격)
        ExtraLife,        // 부활
        SlowProjectile,   // 적 투사체 감속
        BloodThirst,      // 피의 갈증 (처치 시 회복)
    }

    // ─── 애니메이터 해시 (문자열 비교 대신 int 해시) ───
    public static int IDLE = Animator.StringToHash("Idle");  // ②
    public static int MOVE = Animator.StringToHash("Move");
    public static int ATTACK = Animator.StringToHash("Attack");
    public static int DIE = Animator.StringToHash("Die");
    public static int SERVING_IDLE = Animator.StringToHash("ServingIdle");
    public static int SERVING_MOVE = Animator.StringToHash("ServingMove");
    public static int JUMP = Animator.StringToHash("Jump");
    public static int SKILL1 = Animator.StringToHash("Skill1");
    public static int SKILL2 = Animator.StringToHash("Skill2");
    public static int SKILL3 = Animator.StringToHash("Skill3");

    // ─── 장비 슬롯 / 등급 / 스탯 ───
    public enum EEquipSlot { Weapon, Top, Bottom, Bracelet, Ring1, Ring2, Pet1, Pet2 }
    public enum EEquipGrade { Common, Uncommon, Rare, Epic, Legendary }
    public enum EMainStatType { Attack, MaxHP }
    public enum ESubStatType { AttackSpeed, CritChance, CritDamage, DamageResistance }

    // ─── 등급별 색상 ───
    public static Color GetGradeColor(EEquipGrade grade)
    {
        switch (grade)
        {
            case EEquipGrade.Common:    return new Color(0.75f, 0.75f, 0.75f);  // 회색
            case EEquipGrade.Uncommon:  return new Color(0.3f, 0.85f, 0.3f);    // 초록
            case EEquipGrade.Rare:      return new Color(0.3f, 0.5f, 1f);       // 파랑
            case EEquipGrade.Epic:      return new Color(0.7f, 0.3f, 0.9f);     // 보라
            case EEquipGrade.Legendary: return new Color(1f, 0.8f, 0.2f);       // 금색
            default:                    return Color.white;
        }
    }

    // ─── 등급별 배경 Sprite 캐시 (Dictionary O(1) 조회) ───
    private static readonly Dictionary<EEquipGrade, Sprite> _gradeSpriteCache
        = new Dictionary<EEquipGrade, Sprite>();

    public static Sprite GetGradeSprite(EEquipGrade grade)
    {
        if (_gradeSpriteCache.TryGetValue(grade, out Sprite cached) && cached != null)
            return cached;

        string name = grade switch
        {
            EEquipGrade.Common    => "GradeBg_Common",
            EEquipGrade.Uncommon  => "GradeBg_UnCommon",
            EEquipGrade.Rare      => "GradeBg_Rare",
            EEquipGrade.Epic      => "GradeBg_Epic",
            EEquipGrade.Legendary => "GradeBg_Epic",
            _                     => "GradeBg_Common",
        };

        Sprite sprite = IconHelper.GetSprite(name);
        if (sprite != null)
            _gradeSpriteCache[grade] = sprite;

        return sprite;
    }
}
```

#### 코드 해설
| # | 설명 |
|---|------|
| ① | `1 << n` (비트 시프트)로 각 플래그가 고유한 비트 위치를 차지한다. Walk=1(0001), Fly=2(0010), WaterWalk=4(0100)이므로 `Walk | Fly = 3(0011)`처럼 OR 조합으로 복합 능력을 표현할 수 있다 |
| ② | `Animator.StringToHash`로 문자열을 int 해시로 변환하여 static 필드에 캐싱한다. `CrossFade(hash, ...)`로 호출하면 매 프레임 문자열→해시 변환 비용을 제거한다 |

### 설계 의도

> **왜 열거형을 한 파일에 모았는가?** 여러 시스템에서 공유하는 열거형이 파일별로 흩어지면, 새 개발자가 어떤 enum이 어디에 있는지 찾기 어렵다. `Define.cs`에 모아두면 `using static Define;` 한 줄로 모든 열거형에 접근할 수 있고, 프로젝트 전체의 "공용 어휘"를 한눈에 파악할 수 있다.

> **왜 비트 플래그인가?** `ETilePassFlag`를 일반 enum으로 선언하면 "비행하면서 수상보행"같은 복합 능력을 표현할 수 없다. `[Flags]`와 비트 연산을 사용하면 `Walk | Fly | WaterWalk`처럼 OR 조합으로 능력을 자유롭게 조합할 수 있고, 판정 시 `(flags & requiredFlag) != 0`으로 O(1) 체크가 가능하다.

> **왜 해시 캐싱이 중요한가?** `Animator.CrossFade("Idle", 0.1f)`처럼 문자열을 직접 쓰면, Unity 내부에서 매번 해시 변환이 발생한다. 이 게임은 플레이어와 다수의 몬스터가 매 프레임 애니메이션 상태를 체크하므로, 미리 `StringToHash()`로 캐싱해두면 **누적 CPU 비용을 크게 절약**할 수 있다.

- **중앙 집중**: 모든 열거형을 한 파일에 모아 `using static Define;`으로 간결하게 사용
- **비트 플래그**: `ETilePassFlag`를 `[Flags]`로 선언하여 비트 OR 조합으로 복합 통과 능력 표현
- **해시 캐싱**: `Animator.StringToHash()`를 static 필드로 캐싱하여 매 프레임 문자열 비교 회피
- **Sprite 캐시**: `_gradeSpriteCache` Dictionary로 등급 배경 Sprite를 O(1) 조회

## ExpManager - 경험치 / 레벨 관리

> `Assets/Scripts/Manager/ExpManager.cs`

> **왜 레벨업 공식이 `50 + (level × 30)`인가?** 선형 스케일링(`50 + level×30`)은 레벨이 오를수록 필요 경험치가 일정하게 증가한다. 지수 스케일링(`baseExp × 1.5^level`)은 후반부에 급격히 올라가 플레이어가 성장 정체를 느끼기 쉽다. 선형 스케일링은 **매 레벨마다 "조금만 더 모으면 된다"는 느낌**을 유지하여, 짧은 세션의 모바일 게임에 적합하다.

```csharp
public class ExpManager : Singleton<ExpManager>
{
    private int _level = 1;
    private int _currentExp = 0;
    private int _maxLevel = 10;

    public int Level => _level;
    public int CurrentExp => _currentExp;
    public int ExpToNextLevel => 50 + (_level * 30);          // ① 선형 스케일링

    public System.Action<int> OnLevelUp;
    public System.Action<int, int> OnExpChanged;

    public void SetMaxLevel(int level)
        => _maxLevel = Mathf.Max(_maxLevel, level);           // ② 최대치만 갱신

    public void AddExp(int amount)
    {
        // 빠른 성장: 경험치 1.5배
        PlayerUpgrade upgrade = FindAnyObjectByType<PlayerUpgrade>();
        if (upgrade != null && upgrade.HasFastGrowth)
            amount = (int)(amount * 1.5f);                     // ③

        _currentExp += amount;
        while (_currentExp >= ExpToNextLevel && _level < _maxLevel)  // ④
        {
            _currentExp -= ExpToNextLevel;
            _level++;
            AudioManager.Instance.PlayLevelUp();
            OnLevelUp?.Invoke(_level);

            GameObject playerObj = GameObject.FindWithTag("Player");
            if (playerObj != null)
                LevelUpTextSpawner.Spawn(playerObj.transform.position + Vector3.up * 2f);
        }

        if (_level >= _maxLevel)
            _currentExp = 0;                                   // ⑤

        OnExpChanged?.Invoke(_currentExp, ExpToNextLevel);
    }
}
```

#### 코드 해설
| # | 설명 |
|---|------|
| ① | `50 + (level × 30)` 공식으로 레벨 1에서 80, 레벨 5에서 200, 레벨 10에서 350의 필요 경험치를 만든다. 선형 증가로 후반부 성장 정체를 방지한다 |
| ② | `Mathf.Max`로 현재 값보다 높을 때만 갱신한다. FastGrowth 업그레이드로 `SetMaxLevel(15)`를 호출하면 기존 10에서 15로 상향된다 |
| ③ | FastGrowth 보유 시 경험치를 1.5배로 증폭한다. 정수 캐스팅으로 소수점은 버린다 |
| ④ | `while` 루프로 한 번에 경험치가 많이 들어와도 연속 레벨업을 처리한다. 예: 200 경험치가 한번에 들어오면 2레벨 연속 상승 가능 |
| ⑤ | 최대 레벨 도달 시 잔여 경험치를 0으로 리셋하여 경험치 바가 비어있는 상태를 유지한다 |

### 레벨업 데이터 흐름

```
몬스터 사망 → ExpOrb 수집
    │
    ▼
ExpManager.AddExp(amount)
    │  FastGrowth? → amount × 1.5
    ▼
_currentExp += amount
    │
    ├── exp < ExpToNextLevel → OnExpChanged (UI 바 갱신)
    │
    └── exp >= ExpToNextLevel (while 루프)
         ├── _level++
         ├── AudioManager.PlayLevelUp()
         ├── OnLevelUp → UI_LevelUp.Show()
         └── LevelUpTextSpawner.Spawn()
```

## GoldManager - 골드 재화 관리

> `Assets/Scripts/Manager/GoldManager.cs`

> **왜 SaveManager.Save()를 매 골드 변동마다 호출하는가?** 골드는 스테이지를 넘어 누적되는 영구 재화다. 앱이 갑자기 종료되거나 크래시가 발생해도 **마지막 골드 변동 시점의 상태가 보존**되도록, AddGold/SpendGold 호출 시마다 즉시 저장한다. 배치 저장(예: 스테이지 종료 시)보다 빈번하지만, JSON 직렬화 비용이 미미하므로 안전성을 우선한다.

```csharp
public class GoldManager : Singleton<GoldManager>
{
    private int _gold;
    public int Gold => _gold;
    public event Action OnGoldChanged;

    public void AddGold(int amount)                           // ①
    {
        _gold += amount;
        OnGoldChanged?.Invoke();
        SaveManager.Save();
    }

    public bool SpendGold(int amount)                         // ②
    {
        if (_gold < amount) return false;
        _gold -= amount;
        OnGoldChanged?.Invoke();
        SaveManager.Save();
        return true;
    }

    public void SetGold(int amount)                           // ③
    {
        _gold = amount;
        OnGoldChanged?.Invoke();
    }
}
```

#### 코드 해설
| # | 설명 |
|---|------|
| ① | 골드를 추가하고 이벤트를 발행하여 UI를 갱신한 뒤, 즉시 저장한다. 몬스터 드롭, 스테이지 클리어 보상에서 호출된다 |
| ② | 골드가 부족하면 `false`를 반환하여 소비를 거부한다. 호출부에서 반환값으로 구매 가능 여부를 판단할 수 있다 |
| ③ | SaveManager.Load() 시 저장된 골드를 직접 설정한다. Save를 호출하지 않아 로드 시 불필요한 재저장을 방지한다 |

### 골드 흐름도

```
몬스터 사망 → GoldOrb 수집
    │
    ▼
GoldManager.AddGold(5~15)
    │
    ├── _gold += amount
    ├── OnGoldChanged?.Invoke()  → UI 골드 텍스트 갱신
    └── SaveManager.Save()       → JSON 파일 저장

스테이지 클리어
    │
    ▼
GoldManager.AddGold(100)  ← _stageGoldReward
```
